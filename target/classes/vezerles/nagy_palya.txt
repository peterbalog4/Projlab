# ═══════════════════════════════════════════════════════════════
#  "Bízz bennem, mérnök leszek" – 4x4-es városi térkép
# ═══════════════════════════════════════════════════════════════
#
#  4x4 csomópont-rács. Az 1-es sor (n01..n31) és az 1-es oszlop (n10..n13)
#  egy 2+2 sávos főutat alkot, kereszt-alakzatban; a két főút az n11
#  csomópontnál (nagy 4-ágú kereszteződés) találkozik. A többi 18 szakasz
#  1+1 sávos.
#
#  ÚJ KONVENCIÓ: A node-pont mindig a kereszteződés/sávköteg KÖZÉPPONTJÁBAN
#  van (nem a bal-felső sarokban, ahogy korábban). Az utak középvonala áthalad
#  a node-pontokon; ennek köszönhetően a 1+1 és 2+2 sávos utak vizuálisan
#  illeszkednek a kereszteződéseknél.
#
#    n00 ── n10 ── n20 ── n30
#     │     ║      │      │
#    n01 ══ n11 ══ n21 ══ n31      <-- 2+2 sávos vízszintes főút
#     │     ║      │      │
#    n02 ── n12 ── n22 ── n32
#     │     ║      │      │
#    n03 ── n13 ── n23 ── n33
#           ↑
#           2+2 sávos függőleges főút (n10..n13)
#
#  Koordináták (NODE = cella középpontja):
#    1+1 cella mérete  = 120 px (sávköteg 120)
#    2+2 cella mérete  = 240 px (sávköteg 240) – mindenhol az 1. sor / 1. oszlop
#                       node-jainál, mert ott legalább egy 2+2 csatlakozik
#    Egységes spacing  = 320 px minden tengelyen. A leg-szűkebb (2+2↔2+2) szakasz
#                       látható-út = 320 - 120 - 120 = 80 px; az 1+1↔2+2-eken 140 px,
#                       az 1+1↔1+1-en 200 px.
#  Maximum kiterjedés: 1140 × 1140 px.
# ════════════════════════════════════════════════════════════════

# ─── Csomópontok ──────────────────────────────────────────────────
# Sor 0 (csak 1+1 csatlakozók)
node n00   60    60
node n10  380    60
node n20  700    60
node n30 1020    60
# Sor 1 (vízszintes 2+2 főút)
node n01   60   380
node n11  380   380
node n21  700   380
node n31 1020   380
# Sor 2 (1+1)
node n02   60   700
node n12  380   700
node n22  700   700
node n32 1020   700
# Sor 3 (1+1)
node n03   60  1020
node n13  380  1020
node n23  700  1020
node n33 1020  1020

# ─── Vízszintes utak ──────────────────────────────────────────────
# Sor 0 (1+1)
road rh00 1 1 n00 n10
road rh01 1 1 n10 n20 hid       # HÍD: a felső sor egyik 1+1 szakasza
road rh02 1 1 n20 n30
# Sor 1 (2+2 főút)
road rh10 2 2 n01 n11
road rh11 2 2 n11 n21
road rh12 2 2 n21 n31
# Sor 2 (1+1)
road rh20 1 1 n02 n12
road rh21 1 1 n12 n22
road rh22 1 1 n22 n32
# Sor 3 (1+1)
road rh30 1 1 n03 n13
road rh31 1 1 n13 n23
road rh32 1 1 n23 n33

# ─── Függőleges utak ─────────────────────────────────────────────
# Oszlop 0 (1+1)
road rv00 1 1 n00 n01
road rv01 1 1 n01 n02 alagut    # ALAGÚT: a bal oszlop egyik 1+1 szakasza
road rv02 1 1 n02 n03
# Oszlop 1 (2+2 főút)
road rv10 2 2 n10 n11
road rv11 2 2 n11 n12
road rv12 2 2 n12 n13
# Oszlop 2 (1+1)
road rv20 1 1 n20 n21
road rv21 1 1 n21 n22
road rv22 1 1 n22 n23
# Oszlop 3 (1+1)
road rv30 1 1 n30 n31
road rv31 1 1 n31 n32
road rv32 1 1 n32 n33

# ─── ÖSSZEKÖTTETÉSEK (csomópontonként) ─────────────────────────
# Sor 0 -----------------------------------------------------------
# n00 (bal-felső sarok)
connect rh00 vegA rv00 vegA

# n10 (T-elágazás)
connect rh00 vegB rh01 vegA
connect rh00 vegB rv10 vegA
connect rh01 vegA rv10 vegA

# n20 (T)
connect rh01 vegB rh02 vegA
connect rh01 vegB rv20 vegA
connect rh02 vegA rv20 vegA

# n30 (jobb-felső sarok)
connect rh02 vegB rv30 vegA

# Sor 1 (vízszintes főút) -----------------------------------------
# n01 (T)
connect rv00 vegB rv01 vegA
connect rv00 vegB rh10 vegA
connect rv01 vegA rh10 vegA

# n11 (NAGY 4-ágú kereszteződés; a két 2+2 főút találkozik)
connect rh10 vegB rh11 vegA
connect rh10 vegB rv10 vegB
connect rh10 vegB rv11 vegA
connect rh11 vegA rv10 vegB
connect rh11 vegA rv11 vegA
connect rv10 vegB rv11 vegA

# n21 (4-ágú)
connect rh11 vegB rh12 vegA
connect rh11 vegB rv20 vegB
connect rh11 vegB rv21 vegA
connect rh12 vegA rv20 vegB
connect rh12 vegA rv21 vegA
connect rv20 vegB rv21 vegA

# n31 (T)
connect rh12 vegB rv30 vegB
connect rh12 vegB rv31 vegA
connect rv30 vegB rv31 vegA

# Sor 2 -----------------------------------------------------------
# n02 (T)
connect rv01 vegB rv02 vegA
connect rv01 vegB rh20 vegA
connect rv02 vegA rh20 vegA

# n12 (4-ágú)
connect rh20 vegB rh21 vegA
connect rh20 vegB rv11 vegB
connect rh20 vegB rv12 vegA
connect rh21 vegA rv11 vegB
connect rh21 vegA rv12 vegA
connect rv11 vegB rv12 vegA

# n22 (4-ágú)
connect rh21 vegB rh22 vegA
connect rh21 vegB rv21 vegB
connect rh21 vegB rv22 vegA
connect rh22 vegA rv21 vegB
connect rh22 vegA rv22 vegA
connect rv21 vegB rv22 vegA

# n32 (T)
connect rh22 vegB rv31 vegB
connect rh22 vegB rv32 vegA
connect rv31 vegB rv32 vegA

# Sor 3 -----------------------------------------------------------
# n03 (bal-alsó sarok)
connect rv02 vegB rh30 vegA

# n13 (T)
connect rh30 vegB rh31 vegA
connect rh30 vegB rv12 vegB
connect rh31 vegA rv12 vegB

# n23 (T)
connect rh31 vegB rh32 vegA
connect rh31 vegB rv22 vegB
connect rh32 vegA rv22 vegB

# n33 (jobb-alsó sarok)
connect rh32 vegB rv32 vegB

# ─── Telephely ───────────────────────────────────────────────────
depot d0 n00 5000
